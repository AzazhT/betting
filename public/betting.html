/* ======================================================= EVENT CARD ======================================================= */ function eventCard(event) { const odds = event.odds || {}; const over25 = odds.over25 ?? odds.over ?? null; const under25 = odds.under25 ?? odds.under ?? null; const selected1 = isSelected(event.id, "1"); const selectedX = isSelected(event.id, "X"); const selected2 = isSelected(event.id, "2"); const selectedOver = isSelected(event.id, "over"); const selectedUnder = isSelected(event.id, "under"); const selectedYes = isSelected(event.id, "bttsYes"); const selectedNo = isSelected(event.id, "bttsNo"); return `
🏆 ${escapeHtml(event.league || "Football")}
${escapeHtml(event.home?.name || "Home")} ${ event.live ? `${event.score?.home ?? 0}` : "" }
${escapeHtml(event.away?.name || "Away")} ${ event.live ? `${event.score?.away ?? 0}` : "" }
${ event.live ? "🔴 LIVE" : escapeHtml(formatDate(event.startTime)) }
${oddButton(event, "1", "1", odds["1"], selected1)} ${oddButton(event, "X", "X", odds["X"], selectedX)} ${oddButton(event, "2", "2", odds["2"], selected2)} ${oddButton(event, "Over 2.5", "over", over25, selectedOver)} ${oddButton(event, "Under 2.5", "under", under25, selectedUnder)} ${oddButton(event, "BTTS Yes", "bttsYes", odds.bttsYes, selectedYes)} ${oddButton(event, "BTTS No", "bttsNo", odds.bttsNo, selectedNo)}
`; }
